A modern stack behind a simple experience.

React · Tailwind · Vite
  • Component-based, mobile-first responsive layouts

from fastapi import APIRouter, Depends, File, UploadFile, Query
from app.schemas import PropertyOut, PropertyCreate, Page

router = APIRouter(prefix="/api/properties", tags=["properties"])

@router.get("", response_model=Page[PropertyOut])
async def search(
    city: str | None = Query(None),
    price_min: float | None = Query(None, ge=0),
    price_max: float | None = Query(None, ge=0),
    beds: int | None = Query(None, ge=1),
    page: int = Query(1, ge=1),
    db: Session = Depends(get_db),
):
    """Filtered, paginated property search."""
    return property_service.search(db, city, price_min, price_max, beds, page)

@router.post("", status_code=201)
async def create(
    payload: PropertyCreate,
    images: list[UploadFile] = File(...),
    user = Depends(current_agent),
):
    return property_service.create_with_images(db, payload, images, user)
